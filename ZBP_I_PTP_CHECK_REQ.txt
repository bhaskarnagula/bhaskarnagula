  METHOD get_instance_authorizations.
  ENDMETHOD.

  ENDMETHOD.

  METHOD update.
  ENDMETHOD.

  METHOD delete.
  ENDMETHOD.

  METHOD read.
  ENDMETHOD.

  METHOD lock.
  ENDMETHOD.

  METHOD get_instance_features.
    READ ENTITIES OF zi_ptp_check_req IN LOCAL MODE
        ENTITY zi_ptp_check_req
          ALL FIELDS
           WITH CORRESPONDING #( keys )
         RESULT DATA(lt_CHECK_REQ)
         FAILED failed.

    result =
      VALUE #( FOR ls_CHECK_REQ IN lt_CHECK_REQ
        ( %key = ls_CHECK_REQ-%key
          %features-%action-Setfields = COND #( WHEN ls_CHECK_REQ-PoRef = abap_true
                                                        THEN if_abap_behv=>fc-o-disabled
                                                        ELSE if_abap_behv=>fc-o-enabled )
         ) ).
  ENDMETHOD.

  METHOD get_global_features.
  ENDMETHOD.

  METHOD DefaultForCreate.
  ENDMETHOD.

  METHOD Setfields.
  ENDMETHOD.

  METHOD precheck_create.
  ENDMETHOD.

  METHOD set_defaults.
    READ ENTITIES OF zi_ptp_check_req IN LOCAL MODE
        ENTITY zi_ptp_check_req
          FIELDS ( PoRef DocType )
             WITH CORRESPONDING #( keys )
           RESULT DATA(lt_travels).

    LOOP AT lt_travels ASSIGNING FIELD-SYMBOL(<fs_travel>).
      <fs_travel>-PoRef = 'Y'.
    ENDLOOP.

    MODIFY ENTITIES OF zi_ptp_check_req IN LOCAL MODE
      ENTITY zi_ptp_check_req
        UPDATE FIELDS ( PoRef )
        WITH VALUE #( FOR travel IN lt_travels  (
                           %tky      = travel-%tky
                           PoRef  = travel-PoRef ) ).
  ENDMETHOD.

  METHOD validate_data.
  ENDMETHOD.

ENDCLASS.
